def plotar_grafo_rdf(grafos, titulos=None, titulo_geral="Visualização de Grafos Noösphera"):
    """
    Função agnóstica e polimórfica de plotagem.
    Aceita um único grafo rdflib/networkx ou uma lista deles, renderizando-os
    em subplots horizontais limpos e sem poluição visual.
    Simplifica as URIs automaticamente para respeitar o layout Soft Paper.
    """
    import networkx as nx
    import matplotlib.pyplot as plt
    from rdflib import Graph
    
    # Normaliza a entrada para sempre tratar como lista de grafos
    if not isinstance(grafos, list):
        lista_grafos = [grafos]
        lista_titulos = [titulos] if titulos else ["Grafo Noösphera"]
    else:
        lista_grafos = grafos
        lista_titulos = titulos if titulos else [f"Estado {i}" for i in range(len(grafos))]
        
    num_grafos = len(lista_grafos)
    fig, axes = plt.subplots(1, num_grafos, figsize=(12 * num_grafos, 4), squeeze=False)
    
    for idx, (grafo_bruto, sub_titulo) in enumerate(zip(lista_grafos, lista_titulos)):
        ax = axes[0, idx]
        G = nx.DiGraph()
        
        # Se for um grafo nativo do rdflib, extrai e converte as triplas
        if isinstance(grafo_bruto, Graph):
            for sujeito, predicado, objeto in grafo_bruto:
                s_label = sujeito.split('#')[-1] if '#' in sujeito else str(sujeito)
                p_label = predicado.split('#')[-1] if '#' in predicado else str(predicado)
                o_label = objeto.split('#')[-1] if '#' in objeto else str(objeto)
                G.add_edge(s_label, o_label, label=p_label)
        # Se já for um DiGraph do networkx, usa a estrutura diretamente
        elif isinstance(grafo_bruto, nx.DiGraph):
            G = grafo_bruto
            
        if len(G) == 0:
            ax.text(0.5, 0.5, "Grafo Vazio (Fronteira Seca)", ha='center', va='center', fontsize=10, style='italic')
            ax.set_title(sub_titulo, fontsize=12)
            ax.axis('off')
            continue
            
        # Configuração estável do layout físico dos nós
        pos = nx.spring_layout(G, seed=42)
        
        # Desenha a malha topológica básica (Nós e Arestas dirigidas)
        nx.draw(G, pos, ax=ax, with_labels=True, node_color='#4A90E2',
                node_size=2000, font_size=8, font_weight='bold',
                font_color='black', edge_color='gray', arrows=True)
        
        # Injeta os predicados/rótulos em vermelho para auditoria visual rápida
        edge_labels = nx.get_edge_attributes(G, 'label')
        if edge_labels:
            nx.draw_networkx_edge_labels(G, pos, ax=ax, edge_labels=edge_labels, font_color='red', font_size=7)
            
        ax.set_title(sub_titulo, fontsize=12)
        ax.margins(0.2)
        
    plt.suptitle(titulo_geral, fontsize=14, weight='bold')
    plt.tight_layout()
    plt.show()

def extrair_identidade(grafo, K):
    """
    id: K x T -> I
    Mapeamento axiomático que extrai o núcleo identitário invariante de K.
    Garante falha estrita (AssertionError) se a identidade não for unívoca.
    """
    valores = list(grafo.objects(subject=K, predicate=noo.idAbsoluto))
    assert len(valores) == 1, f"❌ Erro Ontológico: id(K) indefinido ou múltiplo para {K}."
    return valores[0]

def obter_representacoes(grafo, K):
    """
    rep: K x T -> P(R)
    Retorna o conjunto passivo de todas as manifestações ativas de K no estado t.
    """
    return set(grafo.objects(subject=K, predicate=noo.manifesta))

def transformar_representacao(grafo, K, r_antiga, r_nova, tipo_nova=noo.Representation):
    """
    update: <i, {r_antiga}> -> <i, {r_nova}>
    Operação atômica e opaca de transição de estado no Runtime.
    Centraliza a mutação usando o verbo homologado 'manifesta'.
    """
    from rdflib import RDF
    if r_antiga is not None:
        grafo.remove((K, noo.manifesta, r_antiga))
    if r_nova is not None:
        grafo.add((r_nova, RDF.type, tipo_nova))
        grafo.add((K, noo.manifesta, r_nova))

def verifica_subgrafo_por_permutacao(g_pequeno, g_grande):
    """
    Mapeamento de inclusão (Embedding estrutural: A ↪ C).
    Verifica de forma exaustiva se g_pequeno está preservado em g_grande.
    """
    if len(g_pequeno.nodes) > len(g_grande.nodes):
        return False
    nodes_peq = list(g_pequeno.nodes)
    nodes_grande = list(g_grande.nodes)
    
    for subset in combinations(nodes_grande, len(nodes_peq)):
        for perm in permutations(subset):
            mapping = dict(zip(nodes_peq, perm))
            ok = True
            for u, v in g_pequeno.edges():
                if not g_grande.has_edge(mapping[u], mapping[v]):
                    ok = False
                    break
            if ok:
                return True
    return False
